async function customerRequest(title = 'درخواست واقعی بررسی پروژه و ساخت ابزار فارسی') {
    const values = { email: `management-ui-${randomBytes(6).toString('hex')}@example.invalid`, password: randomBytes(20).toString('hex'), name: 'مشتری آزمون مدیریت' };
    const r = await api('/api/auth/sign-up/email', '', { ...values, passwordConfirmation: values.password });
    expect(r.status).toBe(200);
    const cookie = cookieOf(r);
    const req = await api('/api/requests', cookie, { operationKey: randomUUID(), title, category: 'DASHBOARD', description: 'برای مدیریت اطلاعات و سفارش‌های کارگاه به یک پنل خوانا و فارسی نیاز دارم.', budgetPreference: 'UNDECIDED' });
    expect(req.status).toBe(201);
    return { ...values, cookie, id: (await req.json()).id };
}

async function acceptedProject(context) {
    const owner = await customerRequest();
    for (const [expectedVersion, toStatus] of [[0, 'REVIEWING'], [1, 'ACCEPTED']]) {
        const r = await context.request.post(`/api/admin/requests/${owner.id}/review`, { headers: { origin }, data: { operationKey: randomUUID(), expectedVersion, toStatus } });
        expect(r.status()).toBe(200);
    }
    const r = await context.request.post(`/api/admin/requests/${owner.id}/convert`, { headers: { origin }, data: { operationKey: randomUUID(), expectedVersion: 2 } });
    expect(r.status()).toBe(200);
    return { ...owner, projectId: (await r.json()).result.id };
}
async function committed(page) { await expect(page.locator('.management-result')).toBeVisible(); }
async function send(page, action, confirm = true) {
    const loaded = page.waitForEvent('domcontentloaded');
    await page.locator(`#management-${action}`).locator('button[type="submit"]').click();
    if (confirm) {
        const dialog = page.getByRole('dialog');
        await expect(dialog).toBeVisible();
        await dialog.getByRole('button', { name: 'تأیید و ثبت', exact: true }).click();
    }
    await loaded;
    await committed(page);
}

scenario('a forced conversion-history failure has no partial project and exact retry succeeds', async ({ context, page }) => {
    await adminLogin(context);
    const owner = await customerRequest();
    for (const [expectedVersion, toStatus] of [[0, 'REVIEWING'], [1, 'ACCEPTED']])
        expect((await context.request.post(`/api/admin/requests/${owner.id}/review`, { headers: { origin }, data: { operationKey: randomUUID(), expectedVersion, toStatus } })).status()).toBe(200);
    await page.goto(`/admin/requests/${owner.id}`);
    const rows = await db.projectStatusHistory.findMany({ select: { id: true } });
    for (const row of rows)
        if (!/^[a-zA-Z0-9-]{20,36}$/.test(row.id))
            throw Error('Unexpected fixture identifier');
    const check = rows.length ? `id IN (${rows.map(r => `'${r.id}'`).join(',')})` : 'id <> id';
    await db.$executeRawUnsafe(`ALTER TABLE ProjectStatusHistory ADD CONSTRAINT management_browser_failure CHECK (${check})`);
    try {
        await page.locator('#management-convert button[type="submit"]').click();
        await page.getByRole('dialog').getByRole('button', { name: 'تأیید و ثبت', exact: true }).click();
        await expect(page.locator('#management-convert .form-feedback')).toContainText('سرویس مدیریت موقتاً');
        expect(await db.project.count({ where: { requestId: owner.id } })).toBe(0);
        expect((await db.projectRequest.findUnique({ where: { id: owner.id } })).status).toBe('ACCEPTED');
    }
    finally {
        await db.$executeRawUnsafe('ALTER TABLE ProjectStatusHistory DROP CHECK management_browser_failure');
    }
    await send(page, 'convert');
    expect(await db.project.count({ where: { requestId: owner.id } })).toBe(1);
});
